   408→function OrderCard({
   409→  order,
   410→  onAccept,
   411→  onDecline,
   412→}: {
   413→  order: OrderDTO
   414→  onAccept: () => void
   415→  onDecline: () => void
   416→}) {
   417→  const left = timeLeft(order.expiresAt)
   418→  const total = Math.max(
   419→    1,
   420→    Math.round((+new Date(order.expiresAt) - +new Date(order.createdAt)) / 1000)
   421→  )
   422→  const leftPct = Math.min(100, Math.round((left / total) * 100))
   423→  const watching = order.watchingDrivers ?? 0
   424→
   425→  return (
   426→    <motion.article
   427→      layout
   428→      initial={{ opacity: 0, y: 14, scale: 0.98 }}
   429→      animate={{ opacity: 1, y: 0, scale: 1 }}
   430→      exit={{ opacity: 0, scale: 0.96, transition: { duration: 0.18 } }}
   431→      transition={{ type: 'spring', stiffness: 320, damping: 28 }}
   432→      className="order-flash group rounded-xl border border-zinc-700/80 bg-zinc-950/60 p-4 transition-all duration-200 hover:-translate-y-0.5 hover:border-emerald-500/40 hover:shadow-lg hover:shadow-emerald-500/10"
   433→    >
   434→      <div className="flex items-start justify-between gap-3">
   435→        <div className="min-w-0">
   436→          <div className="flex min-w-0 flex-wrap items-center gap-2 text-[11px] font-semibold uppercase tracking-wide text-zinc-500">
   437→            <span className="rounded bg-zinc-800 px-1.5 py-0.5 text-zinc-300">{order.tariff.name}</span>
   438→            {order.surge > 1.05 && (
   439→              <span className="rounded bg-amber-400/15 px-1.5 py-0.5 text-amber-300">
   440→                спрос ×{order.surge.toFixed(1)}
   441→              </span>
   442→            )}
   443→            <span
   444→              className={cn(
   445→                'inline-flex items-center gap-1 rounded px-1.5 py-0.5',
   446→                watching >= 15
   447→                  ? 'bg-rose-500/15 text-rose-300'
   448→                  : watching >= 8
   449→                    ? 'bg-amber-400/15 text-amber-300'
   450→                    : 'bg-emerald-500/15 text-emerald-300',
   451→              )}
   452→            >
   453→              <Eye className="h-3 w-3" aria-hidden="true" />
   454→              {watching > 0 &&
   455→                `${watching} ${plural(watching, 'водитель видит', 'водителя видят', 'водителей видят')}`}
   456→            </span>
   457→          </div>
   458→          <p className="mt-1.5 flex items-center gap-1 truncate text-sm font-semibold text-zinc-200">
   459→            <MapPin className="h-3.5 w-3.5 shrink-0 text-emerald-400" />
   460→            {order.from.label}
   461→            <span className="ml-1 shrink-0 text-[11px] font-medium normal-case text-zinc-600">
   462→              {order.distanceKm} км · {order.durationMin} мин
   463→            </span>
   464→          </p>
   465→          <p className="flex items-center gap-1 truncate text-sm text-zinc-400">
   466→            <MapPin className="h-3.5 w-3.5 shrink-0 text-amber-400" />
   467→            {order.to.label}
   468→          </p>
   469→        </div>
   470→        <div className="shrink-0 text-right">
   471→          <p className="text-2xl font-black tabular-nums leading-none text-emerald-400">
   472→            {rub(order.driverPayout)}
   473→          </p>
