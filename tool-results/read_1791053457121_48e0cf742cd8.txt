   355→/** Определяем зону по строке адреса; если не нашли — случайная зона */

   357→  const norm = label.trim().toLowerCase()
   358→  if (norm.length === 0) return randomAddress()
   359→  for (const z of ZONES) {
   360→    const fixed = ADDRESSES[z.id].fixed
   361→    if (fixed) {
   362→      for (const f of fixed) {
   363→        if (f.toLowerCase() === norm) return { label, zoneId: z.id }
   364→      }
   365→    }
   366→  }
   367→  if (norm.length >= 4) {
   368→    for (const z of ZONES) {
   369→      for (const s of ADDRESSES[z.id].streets) {
   370→        const sl = s.toLowerCase()
   371→        if (norm.startsWith(sl) || sl.startsWith(norm)) {
   372→          return { label, zoneId: z.id }
   373→        }
   374→      }
   375→    }
   376→  }
   377→  return { label, zoneId: pick(ZONES).id }
   378→}

   402→const TOTAL_WEIGHT = TARIFFS.reduce((s, t) => s + t.weight, 0)
   403→
   404→function pickWeightedTariff(): TariffDef {
   405→  let r = Math.random() * TOTAL_WEIGHT
   406→  for (const t of TARIFFS) {
   407→    r -= t.weight
   408→    if (r <= 0) return t
   409→  }
   410→  return TARIFFS[0]
   411→}
   412→
   413→function resolveTariff(tariffId: string | null | undefined): TariffDef {
   414→  if (typeof tariffId === 'string') {
   415→    const norm = tariffId.trim().toLowerCase().replace('_', '-')
   416→    const found = TARIFFS.find((t) => t.id === norm)
   417→    if (found) return found
   418→  }
   419→  return pickWeightedTariff()
   420→}

   422→/** Цена = (база + руб/км * км + руб/мин * мин) * множитель тарифа * surge, округление до 10 руб */
   423→function computePrice(t: TariffDef, distanceKm: number, durationMin: number, surge: number): number {
   424→  const raw = (t.base + t.perKm * distanceKm + t.perMin * durationMin) * t.multiplier * surge
   425→  return Math.round(raw / 10) * 10
   426→}

   486→/** Геометрическая дистанция между зонами в км (карта 1000x600 нормированная) */
   487→function zoneDistanceKm(a: ZoneState, b: ZoneState): number {
   488→  const dx = a.x - b.x
   489→  const dy = a.y - b.y
   490→  const km = Math.sqrt(dx * dx + dy * dy) * 0.038 * randFloat(0.88, 1.12)
   491→  return round1(clamp(km, 2, 28))
   492→}

   494→function createOrder(): void {
   495→  if (orders.size >= MAX_FEED) return
   496→  const fromZone = pick(ZONES)
   497→  const toZone = pick(ZONES.filter((z) => z.id !== fromZone.id))
   498→  const tariff = pickWeightedTariff()
   499→  const distanceKm =
   500→    tariff.id === 'intercity' ? round1(randFloat(18, 28)) : zoneDistanceKm(fromZone, toZone)
   501→  const durationMin = round1(distanceKm * 2.2 + 4)
   502→  const surge = surgeForZone(fromZone.id)
   503→  const priceTotal = computePrice(tariff, distanceKm, durationMin, surge)
   504→  const driverPayout = Math.round(priceTotal * 0.88)
   505→  const createdAt = new Date()
   506→  // конкуренция за заказ: чем выше спрос в зоне подачи, тем больше водителей рядом видят ленту
   507→  const watchingDrivers = Math.round(
   508→    clamp(6 + fromZone.demand / 7 + randInt(-2, 3), 3, 24),
   509→  )
   510→  const order: OrderDTO = {

   514→    distanceKm,
   515→    durationMin,
   516→    tariff: { id: tariff.id, name: tariff.name, multiplier: round2(tariff.multiplier) },
   517→    priceTotal,
   518→    driverPayout,
   519→    commissionRub: priceTotal - driverPayout,
   520→    surge: round2(surge),
   521→    createdAt: createdAt.toISOString(),
   522→    expiresAt: new Date(createdAt.getTime() + randInt(35, 60) * 1000).toISOString(),
   523→    watchingDrivers,
   524→  }
   525→  orders.set(order.id, order)
   526→  io.to('drivers').emit('orders:new', { order })
   527→  log(

   529→  )

   570→  }

   655→interface RideSession {
   656→  ride: RideDTO
   657→  timer: ReturnType<typeof setInterval> | null
   658→  /** Все сокеты этой поездки (MVP: пассажир + водитель) — получают ride:state */
   659→  sockets: Socket[]

   666→  mvpOrderId: string | null

   668→  noAutoAssign: boolean

   671→function hintFor(session: RideSession, perspective: 'passenger' | 'driver'): string {
   672→  if (session.phase === 'searching') {
   673→    return SEARCH_HINTS[session.hintIndex % SEARCH_HINTS.length]()
   674→  }
   675→  if (session.phase === 'arriving') {
   676→    return perspective === 'driver' ? 'Вы едете к пассажиру' : 'Водитель выехал к вам'
   677→  }
   678→  if (session.phase === 'onboard') {
   679→    return perspective === 'driver'
   680→      ? 'Пассажир в салоне, следуйте к точке назначения'
   681→      : 'Вы в пути'
   682→  }
   683→  return 'Поездка завершена'
   684→}

   686→/** ride:state всем сокетам сессии — каждому своя перспектива в подсказке */
   687→function broadcastRideState(session: RideSession): void {
   688→  for (const s of session.sockets) {
   689→    if (s.disconnected) continue
   690→    const c = clients.get(s.id)
   691→    const perspective: 'passenger' | 'driver' = c?.role === 'driver' ? 'driver' : 'passenger'
   692→    s.emit('ride:state', {
   693→      ride: { ...session.ride, driver: session.ride.driver ? { ...session.ride.driver } : null },
   694→      hint: hintFor(session, perspective),
   695→    })
   696→  }
   697→}

   699→interface LifecycleOpts {
   700→  mvpOrderId?: string | null
   701→  noAutoAssign?: boolean
   702→}
   703→
   704→function startRideLifecycle(
   705→  sockets: Socket[],
   706→  ride: RideDTO,
   707→  durationMin: number,
   708→  opts?: LifecycleOpts,
   709→): void {
   710→  const noAutoAssign = opts?.noAutoAssign ?? false
   711→  // прерываем предыдущие сессии каждого сокета
   712→  for (const s of sockets) {
   713→    const client = clients.get(s.id)
   714→    if (!client) continue
   715→    if (client.session?.timer) {
   716→      clearInterval(client.session.timer)
   717→      log(`поездка ${client.session.ride.id} прервана (новая поездка)`)
   718→    }
   719→  }
   720→  const session: RideSession = {
   721→    ride,
   722→    timer: null,
   723→    sockets: [...sockets],
   724→    phase: ride.status,
   725→    searchTicksLeft: noAutoAssign ? Number.MAX_SAFE_INTEGER : ride.status === 'searching' ? randInt(3, 6) : 0,
   726→    hintIndex: 0,
   727→    arriveEtaStart: ride.etaMin,
   728→    durationMin,
   729→    mvpOrderId: opts?.mvpOrderId ?? null,
   730→    noAutoAssign,
   731→  }
   732→  for (const s of sockets) {
   733→    const client = clients.get(s.id)
   734→    if (client) client.session = session
   735→  }
   736→  broadcastRideState(session)
   737→  session.timer = setInterval(() => {
   738→    try {
   739→      tickRide(session)
   740→    } catch (err) {
   741→      log(`ошибка тика поездки: ${err instanceof Error ? err.message : String(err)}`)
   742→    }
   743→  }, 2000)
   744→}

   746→function tickRide(session: RideSession): void {
   747→  const ride = session.ride
   748→  if (session.phase === 'searching') {
   749→    session.hintIndex += 1
   750→    broadcastRideState(session)
   751→    if (session.noAutoAssign) return // MVP: ждём реального водителя из ленты
   752→    session.searchTicksLeft -= 1
   753→    if (session.searchTicksLeft <= 0) {
   754→      ride.driver = { ...pick(DRIVER_POOL) }
   755→      ride.status = 'arriving'
   756→      ride.progress = 0
   757→      ride.etaMin = randInt(3, 7)
   758→      session.phase = 'arriving'
   759→      session.arriveEtaStart = ride.etaMin
   760→      broadcastRideState(session)
   761→      log(`поездка ${ride.id}: назначен водитель ${ride.driver.name}, ${ride.driver.car}, подача ~${ride.etaMin} мин`)
   762→    }
   763→    return
   764→  }
   765→  if (session.phase === 'arriving') {
   766→    ride.progress = Math.min(0.25, round2(ride.progress + randFloat(0.07, 0.09)))
   767→    ride.etaMin = Math.max(1, Math.ceil(session.arriveEtaStart * (1 - ride.progress / 0.25)))
   768→    if (ride.progress >= 0.25) {
   769→      ride.status = 'onboard'
   770→      ride.etaMin = Math.max(1, Math.round(session.durationMin))
   771→      session.phase = 'onboard'
   772→      log(`поездка ${ride.id}: пассажир в салоне, в пути ${ride.etaMin} мин`)
   773→    }
   774→    broadcastRideState(session)
   775→    return
   776→  }
   777→  if (session.phase === 'onboard') {
   778→    ride.progress = Math.min(1, round2(ride.progress + randFloat(0.033, 0.042)))
   779→    ride.etaMin = Math.max(0, Math.round(session.durationMin * (1 - (ride.progress - 0.25) / 0.75)))
   780→    if (ride.progress >= 1) {
   781→      ride.progress = 1
   782→      ride.etaMin = 0
   783→      ride.status = 'completed'
   784→      session.phase = 'done'
   785→      completeRide(session)
   786→    } else {
   787→      broadcastRideState(session)
   788→    }
   789→    return
   790→  }
   791→  if (session.timer) {
   792→    clearInterval(session.timer)
   793→    session.timer = null
   794→  }
   795→}
   796→
   797→function completeRide(session: RideSession): void {
   798→  const ride = session.ride
   799→  if (session.timer) {
   800→    clearInterval(session.timer)
   801→    session.timer = null
   802→  }
   803→  const price = ride.priceTotal
   804→  const payout = Math.round(price * 0.88)
   805→  const receipt: ReceiptDTO = {
   806→    price,
   807→    driverPayout: payout,
   808→    commissionRub: price - payout,
   809→    tip: 0,
   810→  }
   811→  for (const s of session.sockets) {
   812→    if (s.disconnected) continue
   813→    s.emit('ride:completed', {
   814→      ride: { ...ride, driver: ride.driver ? { ...ride.driver } : null },
   815→      receipt,
   816→    })
   817→  }
   818→  stats.completedToday += 1
   819→  stats.grossTodayRub += price
   820→  log(`поездка ${ride.id}: завершена, ${price} руб (выплата водителю ${payout})`)
   821→  // заработок обновляем у сокета-водителя этой сессии (если он есть — MVP-режим)
   822→  for (const s of session.sockets) {
   823→    const c = clients.get(s.id)
   824→    if (c && c.role === 'driver' && c.session === session && c.earnings) {
   825→      c.earnings.gross += price
   826→      c.earnings.payout += payout
   827→      c.earnings.commission += price - payout
   828→      c.earnings.rides += 1
   829→      c.earnings.hoursOnline = round1(c.earnings.hoursOnline + 0.1)
   830→      c.week = syncWeekToday(c.week, c.earnings.gross)
   831→      c.lastRides.unshift({ ...ride, driver: ride.driver ? { ...ride.driver } : null })
   832→      c.lastRides = c.lastRides.slice(0, 5)
   833→      s.emit('driver:earnings', {
   834→        today: { ...c.earnings },
   835→        week: c.week.map((d) => ({ ...d })),
   836→        lastRides: c.lastRides.map((r) => ({ ...r, driver: r.driver ? { ...r.driver } : null })),
   837→      })
   838→      break
   839→    }
   840→  }
   841→}

   870→// ============================== История дохода за неделю ==============================
   871→
   872→const WEEK_DAYS = ['Вс', 'Пн', 'Вт', 'Ср', 'Чт', 'Пт', 'Сб'] as const // getDay(): 0=Вс
   873→
   874→/**
   875→ * 7 дней истории: 6 прошлых дней — правдоподобный оборот, сегодня — живые данные смены.
   876→ * Пятница/суббота обычно сильнее (вечерний спрос), середина — ровнее.
   877→ */
   878→function makeWeekHistory(todayGross: number): WeekDayDTO[] {
   879→  const today = new Date()
   880→  const days: WeekDayDTO[] = []
   881→  for (let i = 6; i >= 0; i -= 1) {
   882→    const d = new Date(today.getTime() - i * 86400000)
   883→    const dow = d.getDay()
   884→    const isToday = i === 0
   885→    let gross: number
   886→    if (isToday) {
   887→      gross = todayGross
   888→    } else {
   889→      // база 9..14 тыс, пятница/суббота бонусом
   890→      gross = randInt(9, 14) * 1000 + randInt(0, 900)
   891→      if (dow === 5 || dow === 6) gross += randInt(1200, 2600)
   892→    }
   893→    days.push({
   894→      day: WEEK_DAYS[dow],
   895→      gross,
   896→      payout: Math.round(gross * 0.88),
   897→      yandexPayout: Math.round(gross * 0.6),
   898→      isToday,
   899→    })
   900→  }
   901→  return days
   902→}

   905→function syncWeekToday(week: WeekDayDTO[], gross: number): WeekDayDTO[] {
   906→  return week.map((d) =>
   907→    d.isToday
   908→      ? { ...d, gross, payout: Math.round(gross * 0.88), yandexPayout: Math.round(gross * 0.6) }
   909→      : d,
   910→  )
   911→}
