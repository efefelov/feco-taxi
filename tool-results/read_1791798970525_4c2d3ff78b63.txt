    40→const NB = { style: BorderStyle.NONE, size: 0, color: "FFFFFF" };
    41→function makeTableC({ caption, headers, rows, widths, aligns, headerSize = 20, cellSize = 20 }) {
    42→  const t = P.table;
    43→  const out = [];
    44→  if (caption) {
    45→    out.push(new Paragraph({
    46→      keepNext: true, spacing: { before: 200, after: 80, line: 312 },
    47→      children: [new TextRun({ text: caption, bold: true, size: 21, color: P.secondary, font: CYR_BODY })],
    48→    }));
    49→  }
    50→  const normCell = (cell) => {
    51→    if (typeof cell === "string") return [{ t: cell }];
    52→    if (Array.isArray(cell)) return cell.map((x) => (typeof x === "string" ? { t: x } : x));
    53→    return [cell];
    54→  };
    55→  const mkCell = (cell, i, isHeader, rowIdx) => new TableCell({
    56→    width: { size: widths ? widths[i] : Math.floor(100 / headers.length), type: WidthType.PERCENTAGE },
    57→    margins: { top: 70, bottom: 70, left: 110, right: 110 },
    58→    shading: isHeader
    59→      ? { type: ShadingType.CLEAR, fill: t.headerBg }
    60→      : (rowIdx % 2 === 1 ? { type: ShadingType.CLEAR, fill: t.surface } : undefined),
    61→    children: [new Paragraph({
    62→      alignment: aligns && aligns[i] === "r" ? AlignmentType.RIGHT : AlignmentType.LEFT,
    63→      spacing: { line: 276 },
    64→      children: normCell(cell).map((r) => new TextRun({
    65→        text: r.t, bold: isHeader || !!r.b, size: isHeader ? headerSize : cellSize,
    66→        color: isHeader ? t.headerText : (r.c || C_BODY), font: CYR_BODY,
    67→      })),
    68→    })],
    69→  });
    70→  out.push(new Table({
    71→    width: { size: 100, type: WidthType.PERCENTAGE },
    72→    layout: TableLayoutType.FIXED,
    73→    borders: {
    74→      top: { style: BorderStyle.SINGLE, size: 8, color: t.accentLine },
    75→      bottom: { style: BorderStyle.SINGLE, size: 8, color: t.accentLine },
    76→      left: NB, right: NB,
    77→      insideHorizontal: { style: BorderStyle.SINGLE, size: 2, color: t.innerLine },
    78→      insideVertical: NB,
    79→    },
    80→    rows: [
    81→      new TableRow({ tableHeader: true, cantSplit: true, children: headers.map((x, i) => mkCell(x, i, true, 0)) }),
    82→      ...rows.map((r, ri) => new TableRow({ cantSplit: true, children: r.map((x, i) => mkCell(x, i, false, ri)) })),
    83→    ],
    84→  }));
    85→  out.push(new Paragraph({ spacing: { after: 120 }, children: [] }));
    86→  return out;
    87→}
