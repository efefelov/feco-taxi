    38→const NB = { style: BorderStyle.NONE, size: 0, color: "FFFFFF" };
    39→function makeTableC({ caption, headers, rows, widths, aligns, headerSize = 20, cellSize = 20 }) {
    40→  const t = P.table;
    41→  const out = [];
    42→  if (caption) {
    43→    out.push(new Paragraph({
    44→      keepNext: true, spacing: { before: 200, after: 80, line: 312 },
    45→      children: [new TextRun({ text: caption, bold: true, size: 21, color: P.secondary, font: CYR_BODY })],
    46→    }));
    47→  }
    48→  const normCell = (cell) => {
    49→    if (typeof cell === "string") return [{ t: cell }];
    50→    if (Array.isArray(cell)) return cell.map((x) => (typeof x === "string" ? { t: x } : x));
    51→    return [cell];
    52→  };
    53→  const mkCell = (cell, i, isHeader, rowIdx) => new TableCell({
    54→    width: { size: widths ? widths[i] : Math.floor(100 / headers.length), type: WidthType.PERCENTAGE },
    55→    margins: { top: 70, bottom: 70, left: 110, right: 110 },
    56→    shading: isHeader
    57→      ? { type: ShadingType.CLEAR, fill: t.headerBg }
    58→      : (rowIdx % 2 === 1 ? { type: ShadingType.CLEAR, fill: t.surface } : undefined),
    59→    children: [new Paragraph({
    60→      alignment: aligns && aligns[i] === "r" ? AlignmentType.RIGHT : AlignmentType.LEFT,
    61→      spacing: { line: 276 },
    62→      children: normCell(cell).map((r) => new TextRun({
    63→        text: r.t, bold: isHeader || !!r.b, size: isHeader ? headerSize : cellSize,
    64→        color: isHeader ? t.headerText : (r.c || C_BODY), font: CYR_BODY,
    65→      })),
    66→    })],
    67→  });
    68→  out.push(new Table({
    69→    width: { size: 100, type: WidthType.PERCENTAGE },
    70→    layout: TableLayoutType.FIXED,
    71→    borders: {
    72→      top: { style: BorderStyle.SINGLE, size: 8, color: t.accentLine },
    73→      bottom: { style: BorderStyle.SINGLE, size: 8, color: t.accentLine },
    74→      left: NB, right: NB,
    75→      insideHorizontal: { style: BorderStyle.SINGLE, size: 2, color: t.innerLine },
    76→      insideVertical: NB,
    77→    },
    78→    rows: [
    79→      new TableRow({ tableHeader: true, cantSplit: true, children: headers.map((x, i) => mkCell(x, i, true, 0)) }),
    80→      ...rows.map((r, ri) => new TableRow({ cantSplit: true, children: r.map((x, i) => mkCell(x, i, false, ri)) })),
    81→    ],
    82→  }));
    83→  out.push(new Paragraph({ spacing: { after: 120 }, children: [] }));
    84→  return out;
    85→}
