    64→function hash01(s: string, salt = 0): number {
    65→  let h = 2166136261 ^ salt
    66→  for (let i = 0; i < s.length; i++) {
    67→    h ^= s.charCodeAt(i)
    68→    h = Math.imul(h, 16777619)
    69→  }
    70→  return ((h >>> 0) % 100000) / 100000
    71→}
