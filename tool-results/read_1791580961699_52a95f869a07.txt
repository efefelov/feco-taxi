    28→    28→const NB = { style: BorderStyle.NONE, size: 0, color: "FFFFFF" };
    29→    29→function makeTableC({ caption, headers, rows, widths, aligns, headerSize = 20, cellSize = 20 }) {
    30→    30→  const t = P.table;
    31→    31→  const out = [];
    32→    32→  if (caption) {
    33→    33→    out.push(new Paragraph({
    34→    34→      keepNext: true, spacing: { before: 200, after: 80, line: 312 },
    35→    35→      children: [new TextRun({ text: caption, bold: true, size: 21, color: P.secondary, font: CYR_BODY })],
    36→    36→    }));
    37→    37→  }
    38→    38→  const normCell = (cell) => {
    39→    39→    if (typeof cell === "string") return [{ t: cell }];
    40→    40→    if (Array.isArray(cell)) return cell.map((x) => (typeof x === "string" ? { t: x } : x));
    41→    41→    return [cell];
    42→    42→  };
    43→    43→  const mkCell = (cell, i, isHeader, rowIdx) => new TableCell({
    44→    44→    width: { size: widths ? widths[i] : Math.floor(100 / headers.length), type: WidthType.PERCENTAGE },
    45→    45→    margins: { top: 70, bottom: 70, left: 110, right: 110 },
    46→    46→    shading: isHeader
    47→    47→      ? { type: ShadingType.CLEAR, fill: t.headerBg }
    48→    48→      : (rowIdx % 2 === 1 ? { type: ShadingType.CLEAR, fill: t.surface } : undefined),
    49→    49→    children: [new Paragraph({
    50→    50→      alignment: aligns && aligns[i] === "r" ? AlignmentType.RIGHT : AlignmentType.LEFT,
    51→    51→      spacing: { line: 276 },
    52→    52→      children: normCell(cell).map((r) => new TextRun({
    53→    53→        text: r.t, bold: isHeader || !!r.b, size: isHeader ? headerSize : cellSize,
    54→    54→        color: isHeader ? t.headerText : (r.c || C_BODY), font: CYR_BODY,
    55→    55→      })),
    56→    56→    })],
    57→    57→  });
    58→    58→  out.push(new Table({
    59→    59→    width: { size: 100, type: WidthType.PERCENTAGE },
    60→    60→    layout: TableLayoutType.FIXED,
    61→    61→    borders: {
    62→    62→      top: { style: BorderStyle.SINGLE, size: 8, color: t.accentLine },
    63→    63→      bottom: { style: BorderStyle.SINGLE, size: 8, color: t.accentLine },
    64→    64→      left: NB, right: NB,
    65→    65→      insideHorizontal: { style: BorderStyle.SINGLE, size: 2, color: t.innerLine },
    66→    66→      insideVertical: NB,
    67→    67→    },
    68→    68→    rows: [
    69→    69→      new TableRow({ tableHeader: true, cantSplit: true, children: headers.map((x, i) => mkCell(x, i, true, 0)) }),
    70→    70→      ...rows.map((r, ri) => new TableRow({ cantSplit: true, children: r.map((x, i) => mkCell(x, i, false, ri)) })),
    71→    71→    ],
    72→    72→  }));
    73→    73→  out.push(new Paragraph({ spacing: { after: 120 }, children: [] }));
    74→    74→  return out;
    75→    75→}
