   375→   375→function OrderCard({
   376→   376→  order,
   377→   377→  onAccept,
   378→   378→  onDecline,
   379→   379→}: {

   383→   383→}) {
   384→   384→  const left = timeLeft(order.expiresAt)
   385→   385→  const total = Math.max(
   386→   386→    1,
   387→   387→    Math.round((+new Date(order.expiresAt) - +new Date(order.createdAt)) / 1000)
   388→   388→  )
   389→   389→  const leftPct = Math.min(100, Math.round((left / total) * 100))
   390→   390→  const watching = order.watchingDrivers ?? 0
   391→   391→
   392→   392→  return (
   393→   393→    <motion.article
   394→   394→      layout
   395→   395→      initial={{ opacity: 0, y: 14, scale: 0.98 }}
   396→   396→      animate={{ opacity: 1, y: 0, scale: 1 }}
   397→   397→      exit={{ opacity: 0, scale: 0.96, transition: { duration: 0.18 } }}
   398→   398→      transition={{ type: 'spring', stiffness: 320, damping: 28 }}
   399→   399→      className="order-flash group rounded-xl border border-zinc-700/80 bg-zinc-950/60 p-4 transition-all duration-200 hover:-translate-y-0.5 hover:border-emerald-500/40 hover:shadow-lg hover:shadow-emerald-500/10"
   400→   400→    >
   401→   401→      <div className="flex items-start justify-between gap-3">
