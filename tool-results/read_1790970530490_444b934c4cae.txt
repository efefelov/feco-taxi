   342→/** Определяем зону по строке адреса; если не нашли — случайная зона */
   343→function resolveAddress(label: string): AddressPoint {
   344→  const norm = label.trim().toLowerCase()
   345→  if (norm.length === 0) return randomAddress()
   346→  for (const z of ZONES) {
   347→    const fixed = ADDRESSES[z.id].fixed
   348→    if (fixed) {
   349→      for (const f of fixed) {
   350→        if (f.toLowerCase() === norm) return { label, zoneId: z.id }
   351→      }
   352→    }
   353→  }
   354→  if (norm.length >= 4) {
   355→    for (const z of ZONES) {
   356→      for (const s of ADDRESSES[z.id].streets) {
   357→        const sl = s.toLowerCase()
   358→        if (norm.startsWith(sl) || sl.startsWith(norm)) {
   359→          return { label, zoneId: z.id }
   360→        }
   361→      }
   362→    }
   363→  }
   364→  return { label, zoneId: pick(ZONES).id }
   365→}

   369→interface TariffDef {

   389→const TOTAL_WEIGHT = TARIFFS.reduce((s, t) => s + t.weight, 0)
   390→
   391→function pickWeightedTariff(): TariffDef {
   392→  let r = Math.random() * TOTAL_WEIGHT
   393→  for (const t of TARIFFS) {
   394→    r -= t.weight
   395→    if (r <= 0) return t
   396→  }
   397→  return TARIFFS[0]
   398→}

   400→function resolveTariff(tariffId: string | null | undefined): TariffDef {
   401→  if (typeof tariffId === 'string') {
   402→    const norm = tariffId.trim().toLowerCase().replace('_', '-')
   403→    const found = TARIFFS.find((t) => t.id === norm)
   404→    if (found) return found
   405→  }
   406→  return pickWeightedTariff()
   407→}

   409→/** Цена = (база + руб/км * км + руб/мин * мин) * множитель тарифа * surge, округление до 10 руб */
   410→function computePrice(t: TariffDef, distanceKm: number, durationMin: number, surge: number): number {
   411→  const raw = (t.base + t.perKm * distanceKm + t.perMin * durationMin) * t.multiplier * surge
   412→  return Math.round(raw / 10) * 10
   413→}

   471→/** Геометрическая дистанция между зонами в км (карта 1000x600 нормированная) */
   472→function zoneDistanceKm(a: ZoneState, b: ZoneState): number {
   473→  const dx = a.x - b.x
   474→  const dy = a.y - b.y
   475→  const km = Math.sqrt(dx * dx + dy * dy) * 0.038 * randFloat(0.88, 1.12)
   476→  return round1(clamp(km, 2, 28))
   477→}

   479→function createOrder(): void {
   480→  if (orders.size >= MAX_FEED) return
   481→  const fromZone = pick(ZONES)
   482→  const toZone = pick(ZONES.filter((z) => z.id !== fromZone.id))
   483→  const tariff = pickWeightedTariff()
   484→  const distanceKm =
   485→    tariff.id === 'intercity' ? round1(randFloat(18, 28)) : zoneDistanceKm(fromZone, toZone)
   486→  const durationMin = round1(distanceKm * 2.2 + 4)
   487→  const surge = surgeForZone(fromZone.id)
   488→  const priceTotal = computePrice(tariff, distanceKm, durationMin, surge)
   489→  const driverPayout = Math.round(priceTotal * 0.92)
   490→  const createdAt = new Date()
   491→  const order: OrderDTO = {
   492→    id: nextOrderId(),
   493→    from: pickAddressFor(fromZone),
   494→    to: pickAddressFor(toZone),
   495→    distanceKm,
   496→    durationMin,
   497→    tariff: { id: tariff.id, name: tariff.name, multiplier: round2(tariff.multiplier) },
   498→    priceTotal,
   499→    driverPayout,
   500→    commissionRub: priceTotal - driverPayout,
   501→    surge: round2(surge),
   502→    createdAt: createdAt.toISOString(),
   503→    expiresAt: new Date(createdAt.getTime() + randInt(35, 60) * 1000).toISOString(),
   504→  }
   505→  orders.set(order.id, order)
   506→  io.to('drivers').emit('orders:new', { order })
   507→  log(
   508→    `заказ ${order.id}: ${order.from.label} -> ${order.to.label}, ${order.distanceKm} км, ${order.priceTotal} руб (${order.tariff.id}${order.surge > 1 ? `, surge ${order.surge}` : ''})`,
   509→  )
   510→}

   514→  io.to('drivers').emit('orders:remove', { id, reason })
   515→  return true
   516→}

   518→/** Жизнь заказа 35..60 сек, потом истекает */

   520→  const now = Date.now()
   521→  for (const order of orders.values()) {
   522→    if (now >= Date.parse(order.expiresAt)) {
   523→      removeOrder(order.id, 'expired')
   524→      log(`заказ ${order.id} истёк`)
   525→    }
   526→  }

   570→/** Подсказки фазы поиска водителя (ротация каждые 2 сек) */
   571→const SEARCH_HINTS = [
   572→  () => 'Ищем ближайшего водителя...',
   573→  () => `Найдено ${pick([5, 8, 11, 12, 14, 17, 25, 28])} водителей рядом`,
   574→  () => 'Подбираем машину под ваш тариф...',
   575→  () => 'Согласовываем заказ с водителем...',
   576→  () => 'Осталось совсем немного...',
   577→] as const

   579→interface RideSession {
   580→  ride: RideDTO
   581→  timer: ReturnType<typeof setInterval> | null
   582→  perspective: 'passenger' | 'driver'
   583→  phase: RideStatus | 'done'
   584→  searchTicksLeft: number // тиков по 2 сек до назначения водителя (3..6 => 6..12 сек)
   585→  hintIndex: number
   586→  arriveEtaStart: number
   587→  durationMin: number
   588→}
   589→
   590→function hintFor(session: RideSession): string {
   591→  if (session.phase === 'searching') {
   592→    return SEARCH_HINTS[session.hintIndex % SEARCH_HINTS.length]()
   593→  }
   594→  if (session.phase === 'arriving') {
   595→    return session.perspective === 'driver' ? 'Вы едете к пассажиру' : 'Водитель выехал к вам'
   596→  }
   597→  if (session.phase === 'onboard') {
   598→    return session.perspective === 'driver'
   599→      ? 'Пассажир в салоне, следуйте к точке назначения'
   600→      : 'Вы в пути'
   601→  }
   602→  return 'Поездка завершена'
   603→}
   604→
   605→function emitRideState(socket: Socket, session: RideSession): void {
   606→  socket.emit('ride:state', { ride: session.ride, hint: hintFor(session) })
   607→}

   609→function startRideLifecycle(
   610→  socket: Socket,
   611→  ride: RideDTO,
   612→  perspective: 'passenger' | 'driver',
   613→  durationMin: number,
   614→): void {
   615→  const client = clients.get(socket.id)
   616→  if (!client) return
   617→  if (client.session?.timer) {
   618→    clearInterval(client.session.timer)
   619→    log(`поездка ${client.session.ride.id} прервана (новая поездка)`)
   620→  }
   621→  const session: RideSession = {
   622→    ride,
   623→    timer: null,
   624→    perspective,
   625→    phase: ride.status,
   626→    searchTicksLeft: perspective === 'passenger' ? randInt(3, 6) : 0,
   627→    hintIndex: 0,
   628→    arriveEtaStart: ride.etaMin,
   629→    durationMin,
   630→  }
   631→  client.session = session
   632→  emitRideState(socket, session)
   633→  session.timer = setInterval(() => {
   634→    try {
   635→      tickRide(socket, session)
   636→    } catch (err) {
   637→      log(`ошибка тика поездки: ${err instanceof Error ? err.message : String(err)}`)
   638→    }
   639→  }, 2000)
   640→}
   641→
   642→function tickRide(socket: Socket, session: RideSession): void {
   643→  const ride = session.ride
   644→  if (session.phase === 'searching') {
   645→    session.hintIndex += 1
   646→    emitRideState(socket, session)
   647→    session.searchTicksLeft -= 1
   648→    if (session.searchTicksLeft <= 0) {
   649→      ride.driver = { ...pick(DRIVER_POOL) }
   650→      ride.status = 'arriving'
   651→      ride.progress = 0
   652→      ride.etaMin = randInt(3, 7)
   653→      session.phase = 'arriving'
   654→      session.arriveEtaStart = ride.etaMin
   655→      emitRideState(socket, session)
   656→      log(`поездка ${ride.id}: назначен водитель ${ride.driver.name}, ${ride.driver.car}, подача ~${ride.etaMin} мин`)
   657→    }
   658→    return
   659→  }
   660→  if (session.phase === 'arriving') {
   661→    ride.progress = Math.min(0.25, round2(ride.progress + randFloat(0.07, 0.09)))
   662→    ride.etaMin = Math.max(1, Math.ceil(session.arriveEtaStart * (1 - ride.progress / 0.25)))
   663→    if (ride.progress >= 0.25) {
   664→      ride.status = 'onboard'
   665→      ride.etaMin = Math.max(1, Math.round(session.durationMin))
   666→      session.phase = 'onboard'
   667→      log(`поездка ${ride.id}: пассажир в салоне, в пути ${ride.etaMin} мин`)
   668→    }
   669→    emitRideState(socket, session)
   670→    return
   671→  }
   672→  if (session.phase === 'onboard') {
   673→    ride.progress = Math.min(1, round2(ride.progress + randFloat(0.033, 0.042)))
   674→    ride.etaMin = Math.max(0, Math.round(session.durationMin * (1 - (ride.progress - 0.25) / 0.75)))
   675→    if (ride.progress >= 1) {
   676→      ride.progress = 1
   677→      ride.etaMin = 0
   678→      ride.status = 'completed'
   679→      session.phase = 'done'
   680→      completeRide(socket, session)
   681→    } else {
   682→      emitRideState(socket, session)
   683→    }
   684→    return
   685→  }
   686→  if (session.timer) {
   687→    clearInterval(session.timer)
   688→    session.timer = null
   689→  }
   690→}
   691→
   692→function completeRide(socket: Socket, session: RideSession): void {
   693→  const ride = session.ride
   694→  if (session.timer) {
   695→    clearInterval(session.timer)
   696→    session.timer = null
   697→  }
   698→  const price = ride.priceTotal
   699→  const payout = Math.round(price * 0.92)
   700→  const receipt: ReceiptDTO = {
   701→    price,
   702→    driverPayout: payout,
   703→    commissionRub: price - payout,
   704→    tip: 0,
   705→  }
   706→  socket.emit('ride:completed', {
   707→    ride: { ...ride, driver: ride.driver ? { ...ride.driver } : null },
   708→    receipt,
   709→  })
   710→  stats.completedToday += 1
   711→  stats.grossTodayRub += price
   712→  log(`поездка ${ride.id}: завершена, ${price} руб (выплата водителю ${payout})`)
   713→  const client = clients.get(socket.id)
   714→  if (client && client.role === 'driver' && client.earnings) {
   715→    client.earnings.gross += price
   716→    client.earnings.payout += payout
   717→    client.earnings.commission += price - payout
   718→    client.earnings.rides += 1
   719→    client.earnings.hoursOnline = round1(client.earnings.hoursOnline + 0.1)
   720→    client.lastRides.unshift({ ...ride, driver: ride.driver ? { ...ride.driver } : null })
   721→    client.lastRides = client.lastRides.slice(0, 5)
   722→    socket.emit('driver:earnings', {
   723→      today: { ...client.earnings },
   724→      lastRides: client.lastRides.map((r) => ({ ...r, driver: r.driver ? { ...r.driver } : null })),
   725→    })
   726→  }
   727→}

   730→function makeLastRides(driver: DriverInfo): RideDTO[] {
   731→  const rides: RideDTO[] = []
   732→  for (let i = 0; i < 3; i++) {
   733→    const from = randomAddress()
   734→    const to = randomAddress(from.zoneId)
   735→    const t = pick([TARIFFS[0], TARIFFS[1], TARIFFS[2]])
   736→    const d = round1(randFloat(4, 18))
   737→    const price = computePrice(t, d, round1(d * 2.2 + 4), 1)
   738→    rides.push({
   739→      id: nextRideId(),
   740→      status: 'completed',
   741→      driver: { ...driver },
   742→      from,
   743→      to,
   744→      distanceKm: d,
   745→      etaMin: 0,
   746→      progress: 1,
   747→      priceTotal: price,
   748→      driverPayout: Math.round(price * 0.92),
   749→      tariffName: t.name,
   750→      startedAt: new Date(Date.now() - randInt(30, 360) * 60000).toISOString(),
   751→    })
   752→  }
   753→  return rides.sort((a, b) => (a.startedAt < b.startedAt ? 1 : -1))
   754→}

   859→    safeArg<unknown>((raw) => {
   860→      // Отказ от заказа — БЕЗ штрафа и понижения приоритета (наш дифференциатор)
   861→      const p = isObj(raw)
   862→      const orderId = typeof p.orderId === 'string' ? p.orderId : '(без id)'
   863→      log(`водитель ${socket.id.slice(0, 8)} отказался от заказа ${orderId} — без штрафа`)

   873→      if (client.role !== 'passenger') client.role = 'passenger'
   874→      // частичные/пустые поля -> случайные значения
   875→      const fromRaw = typeof p.from === 'string' && p.from.trim() !== '' ? p.from.trim() : null
   876→      const toRaw = typeof p.to === 'string' && p.to.trim() !== '' ? p.to.trim() : null
   877→      const tariff = resolveTariff(typeof p.tariffId === 'string' ? p.tariffId : null)
   878→      const from = fromRaw ? resolveAddress(fromRaw) : randomAddress()
   879→      const to = toRaw ? resolveAddress(toRaw) : randomAddress(from.zoneId)

   933→      }

   936→        log(`водитель ${socket.id.slice(0, 8)}: заказ ${orderId} отклонён (уже в поездке)`)

   938→      }

   940→      // заказ удалён глобально, orders:remove получает и сам принимающий сокет

   942→      const identity = client.identity ?? makeDriver()
   943→      const ride: RideDTO = {
   944→        id: nextRideId(),
   945→        status: 'arriving',
   946→        driver: { ...identity },

   949→        distanceKm: order.distanceKm,
   950→        etaMin: randInt(3, 7),
   951→        progress: 0,

   970→  })
